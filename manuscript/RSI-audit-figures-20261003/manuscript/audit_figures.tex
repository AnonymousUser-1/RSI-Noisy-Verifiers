\begin{figure*}[t]
  \centering
  \includegraphics[width=\textwidth]{figures/audit_map.pdf}
  \caption{Targeted auditing changes the initial update of the structured rule S.
  We fix $a=0.75$, $b=0.5$, $\theta_0=0$, and the mathematical step size
  $\eta=0.1$. Color and vertical displacement show
  $100[P(-\eta g_{S,q}(0))-1/2]$ in percentage points (pp).
  (a) Response over magnitude $L$ and high-group audit probability $q$.
  The solid curve is the initial zero-change boundary
  $q_c(L)=0.25-0.75/L$, shown only where $q_c\geq0$;
  the dashed line is the uniform sufficient threshold $q=0.25$ of Theorem~4.5.
  (b) Exact parameter slices for $L=3,10,20$.
  The open marker identifies $q_c(10)=0.175$.
  Here $q$ applies only to accepted high-group candidates; the expected fraction
  of the entire pre-audit accepted pool queried at initialization is $B(q)=0.7q$.
  These are deterministic population calculations, not language-model results
  or finite-time demonstrations of asymptotic convergence.}
  \label{fig:audited-initial-map}
\end{figure*}
